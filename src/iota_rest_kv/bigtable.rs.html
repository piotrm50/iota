50/// Provides read access to data ingested by the `iota-data-ingestion`
51/// crate's `KVStoreWorker`.
52///
53/// It retrieves data from BigTableDB.
54///
55/// The client implements a read-only interface and supports the HTTP fallback
56/// mechanism used by
57/// [`HttpKVStore`](iota_storage::http_key_value_store::HttpKVStore).
58#[derive(Clone)]
59pub struct KvStoreClient {
60    /// BigTableDB client.
61    bigtable_client: BigTableClient,
62    /// The representation of the uptime of the service.
63    start_time: Instant,
64}

66impl KvStoreClient {
67    /// Create a new instance of the client.
68    ///
69    /// Internally it instantiates a BigTableDB client.
70    pub async fn new(config: KvStoreConfig) -> Result<Self> {
71        let bigtable_client = if let Some(emulator_host) = config.emulator_host {
72            std::env::set_var("BIGTABLE_EMULATOR_HOST", &emulator_host);
73            BigTableClient::new_local(config.instance_id, config.column_family).await?
74        } else {
75            BigTableClient::new_remote(
76                config.instance_id,
77                true,
78                Some(Duration::from_secs(config.timeout_secs as u64)),
79                "rest".to_string(),
80                config.column_family,
81                None,
82            )
83            .await?
84        };
85
86        Ok(Self {
87            bigtable_client,
88            start_time: Instant::now(),
89        })
90    }

92    /// Builds a [`RowFilter`] that matches only cells whose column qualifier
93    /// equals `column_qualifier` exactly.
94    ///
95    /// Internally the filter is implemented as a regex anchored with `^` and
96    /// `$` to enforce an exact byte match of the provided
97    /// `column_qualifier`, preventing partial or substring matches against
98    /// other columns whose qualifiers happen to contain `column_qualifier`
99    /// as a prefix, suffix, or substring.
100    fn column_qualifier_filter(column_qualifier: &str) -> RowFilter {
101        RowFilter {
102            filter: Some(Filter::ColumnQualifierRegexFilter(
103                format!("^{column_qualifier}$").into_bytes(),
104            )),
105        }
106    }

110        self.start_time.elapsed()

113    /// Gets value as [`Bytes`] from the kv store.
114    ///
115    /// Based on the provided [`Key`] fetch the data from BigTableDB.
116    pub async fn get(&self, key: Key) -> Result<Option<Bytes>, ApiError> {
117        let results = self.get_items(vec![key]).await?;
118        Ok(results.into_iter().next().unwrap_or(None))
119    }

121    /// Gets multiple values as [`Vec`]<[`Option`]<[`Bytes`]>> from the kv
122    /// store.
123    ///
124    /// Based on the provided [`Vec`]<[`Key`]> fetch the data from BigTableDB.
125    /// Returns a vector of the same length and order as the input keys.
126    /// Each entry is `Some(bytes)` if the key was found, or `None` if not
127    /// found.
128    ///
129    /// All keys must be of the same type, otherwise [`ApiError::BadRequest`] is
130    /// returned.
131    pub async fn get_items(&self, keys: Vec<Key>) -> Result<Vec<Option<Bytes>>, ApiError> {
132        if keys.is_empty() {
133            return Ok(Vec::new());
134        }
135
136        // Use the first key to determine the type - all keys should be of the same type
137        match keys.first().expect("emptiness was checked earlier") {
138            Key::Transaction(_) => {
139                let digests = extract_keys(&keys, |k| match k {
140                    Key::Transaction(digest) => Some(*digest),
141                    _ => None,
142                })?;
143
144                let keys = digests.iter().map(|tx| Some(tx.inner().to_vec())).collect();
145
146                self.fetch_from_bigtable(TRANSACTIONS_TABLE, keys, TRANSACTION_COLUMN_QUALIFIER)
147                    .await
148            }
149            Key::TransactionEffects(_) => {
150                let digests = extract_keys(&keys, |k| match k {
151                    Key::TransactionEffects(digest) => Some(*digest),
152                    _ => None,
153                })?;
154
155                let keys = digests.iter().map(|tx| Some(tx.inner().to_vec())).collect();
156
157                self.fetch_from_bigtable(TRANSACTIONS_TABLE, keys, EFFECTS_COLUMN_QUALIFIER)
158                    .await
159            }
160            Key::CheckpointContents(_) => {
161                let seq_nums = extract_keys(&keys, |k| match k {
162                    Key::CheckpointContents(seq_num) => Some(*seq_num),
163                    _ => None,
164                })?;
165
166                let keys = seq_nums
167                    .iter()
168                    .map(|sq| Some(sq.to_be_bytes().to_vec()))
169                    .collect();
170
171                self.fetch_from_bigtable(
172                    CHECKPOINTS_TABLE,
173                    keys,
174                    CHECKPOINT_CONTENTS_COLUMN_QUALIFIER,
175                )
176                .await
177            }
178            Key::CheckpointSummary(_) => {
179                let seq_nums = extract_keys(&keys, |k| match k {
180                    Key::CheckpointSummary(seq_num) => Some(*seq_num),
181                    _ => None,
182                })?;
183
184                let keys = seq_nums
185                    .iter()
186                    .map(|sq| Some(sq.to_be_bytes().to_vec()))
187                    .collect();
188
189                self.fetch_from_bigtable(
190                    CHECKPOINTS_TABLE,
191                    keys,
192                    CHECKPOINT_SUMMARY_COLUMN_QUALIFIER,
193                )
194                .await
195            }
196            Key::CheckpointSummaryByDigest(_) => {
197                let checkpoint_digests = extract_keys(&keys, |k| match k {
198                    Key::CheckpointSummaryByDigest(checkpoint_digest) => Some(*checkpoint_digest),
199                    _ => None,
200                })?;
201
202                let digest_keys = checkpoint_digests
203                    .iter()
204                    .map(|digest| Some(digest.inner().to_vec()))
205                    .collect::<Vec<Option<Vec<u8>>>>();
206
207                self.checkpoint_summary_by_digests(digest_keys).await
208            }
209            Key::TransactionToCheckpoint(_) => {
210                let digests = extract_keys(&keys, |k| match k {
211                    Key::TransactionToCheckpoint(digest) => Some(*digest),
212                    _ => None,
213                })?;
214
215                let keys = digests.iter().map(|tx| Some(tx.inner().to_vec())).collect();
216
217                self.fetch_from_bigtable(TRANSACTIONS_TABLE, keys, TRANSACTION_TO_CHECKPOINT)
218                    .await
219            }
220            Key::ObjectKey(_) => {
221                let object_keys = extract_keys(&keys, |k| match k {
222                    Key::ObjectKey(object_key) => Some(*object_key),
223                    _ => None,
224                })?;
225
226                let keys = object_keys
227                    .iter()
228                    .map(|key| Some(raw_object_key(key)))
229                    .collect();
230
231                self.fetch_from_bigtable(OBJECTS_TABLE, keys, DEFAULT_COLUMN_QUALIFIER)
232                    .await
233            }
234            Key::EventsByTransactionDigest(_) => {
235                let digests = extract_keys(&keys, |k| match k {
236                    Key::EventsByTransactionDigest(digest) => Some(*digest),
237                    _ => None,
238                })?;
239
240                let keys = digests.iter().map(|tx| Some(tx.inner().to_vec())).collect();
241
242                let response = self
243                    .fetch_from_bigtable(TRANSACTIONS_TABLE, keys, EVENTS_COLUMN_QUALIFIER)
244                    .await?;
245
246                Ok(response
247                    .into_iter()
248                    .map(|cell| {
249                        cell.and_then(|bytes| {
250                            match bcs::from_bytes::<Option<TransactionEvents>>(&bytes) {
251                                Ok(None) | Err(_) => None,
252                                Ok(Some(events)) => bcs::to_bytes(&events).map(Bytes::from).ok(),
253                            }
254                        })
255                    })
256                    .collect())
257            }
258        }
259        .map_err(Into::into)
260    }

262    /// Fetch multiple values from a BigTable table with a specific key and
263    /// column qualifier.
264    ///
265    /// Keys wrapped in `Option<Vec<u8>>` allow chaining multiple queries: the
266    /// result from one `fetch_from_bigtable` (which contains `None` for missing
267    /// keys) can be directly passed as input to the next call. `None` keys
268    /// are skipped in the query but preserve their position in the result.
269    ///
270    /// The result's length is guaranteed to match the input `keys` length. Each
271    /// position in the result corresponds to the key at the same position in
272    /// the input. This allows the caller to easily determine which
273    /// requested keys have data:
274    /// - `Some(value)` at index `i` means `key[i]` exists and has data
275    /// - `None` at index `i` means `key[i]` was not found or has no matching
276    ///   data
277    async fn fetch_from_bigtable(
278        &self,
279        table_name: &str,
280        keys: Vec<Option<Vec<u8>>>,
281        column_qualifier: &str,
282    ) -> Result<Vec<Option<Bytes>>, anyhow::Error> {
283        let mut client = self.bigtable_client.clone();
284        // pre-allocate results with None. Matching cells will replace None with
285        // Some(value), and unmatched keys will remain None.
286        let mut results = vec![None; keys.len()];
287
288        let key_to_index = keys
289            .iter()
290            .enumerate()
291            .filter_map(|(index, key)| key.as_ref().map(|k| (k.clone(), index)))
292            .collect::<HashMap<Vec<u8>, usize>>();
293
294        for row in client
295            .multi_get(
296                table_name,
297                key_to_index.keys().cloned().collect(),
298                Some(Self::column_qualifier_filter(column_qualifier)),
299            )
300            .await?
301        {
302            for Cell { name, value } in row.cells {
303                let cell_name = std::str::from_utf8(&name)?;
304                if cell_name == column_qualifier {
305                    if let Some(&index) = key_to_index.get(&row.key) {
306                        results[index] = Some(Bytes::from(value));
307                    }
308                } else {
309                    error!("unexpected column {cell_name:?} in {table_name} table")
310                }
311            }
312        }
313
314        Ok(results)
315    }

317    /// Fetch multiple checkpoint summaries by its checkpoint digest.
318    async fn checkpoint_summary_by_digests(
319        &self,
320        keys: Vec<Option<Vec<u8>>>,
321    ) -> Result<Vec<Option<Bytes>>, anyhow::Error> {
322        let sequence_numbers = self
323            .fetch_from_bigtable(CHECKPOINTS_BY_DIGEST_TABLE, keys, DEFAULT_COLUMN_QUALIFIER)
324            .await?;
325
326        let seq_numbers_keys = sequence_numbers
327            .into_iter()
328            .map(|bytes| bytes.map(|b| b.to_vec()))
329            .collect::<Vec<Option<Vec<u8>>>>();
330
331        self.fetch_from_bigtable(
332            CHECKPOINTS_TABLE,
333            seq_numbers_keys,
334            CHECKPOINT_SUMMARY_COLUMN_QUALIFIER,
335        )
336        .await
337    }

339    /// Performs a single reverse range scan against the objects table and
340    /// returns the bytes of the highest stored version that is **strictly
341    /// less than** the version encoded in [`ObjectRangeKeyBound::end_key`].
342    ///
343    /// Returns `None` if no version below the requested one exists for that
344    /// object (or if the object is not stored at all).
345    pub async fn object_before_version(
346        &self,
347        range: ObjectRangeKeyBound,
348    ) -> Result<Option<Bytes>, anyhow::Error> {
349        if range.is_empty() {
350            return Ok(None);
351        }
352
353        let mut client = self.bigtable_client.clone();
354
355        let reversed = true;
356        let rows_limit = 1;
357        let rows = client
358            .range_scan(
359                OBJECTS_TABLE,
360                Some(range.start_key),
361                Some(range.end_key),
362                rows_limit,
363                reversed,
364                Some(Self::column_qualifier_filter(DEFAULT_COLUMN_QUALIFIER)),
365            )
366            .await?;
367
368        let Some(row) = rows.into_iter().next() else {
369            return Ok(None);
370        };
371
372        let mut value = None;
373        for Cell {
374            name,
375            value: cell_value,
376        } in row.cells
377        {
378            // little optimization, comparing bytes is cheaper than converting it to utf8
379            // string and do string comparison.
380            if name == DEFAULT_COLUMN_QUALIFIER.as_bytes() {
381                value = Some(Bytes::from(cell_value));
382            } else {
383                let cell_name = std::str::from_utf8(&name)?;
384                error!("unexpected column {cell_name:?} in {OBJECTS_TABLE} table");
385            }
386        }
387        Ok(value)
388    }

390    /// Gets multiple objects returning the latest version strictly less than
391    /// the requested version values as [`Vec`]<[`Option`]<[`Bytes`]>> from the
392    /// kv store.
393    ///
394    /// Based on the provided [`ObjectsBeforeVersionRequest`] fetch the data
395    /// from BigTableDB. Returns a vector of the same length and order as
396    /// the input keys. Each entry is `Some(bytes)` if the key was found, or
397    /// `None` if not found.
398    pub async fn objects_before_version(
399        &self,
400        req: ObjectsBeforeVersionRequest,
401    ) -> Result<Vec<Option<Bytes>>, anyhow::Error> {
402        let req = req.into_inner();
403
404        // `multiget_max_items` bounds the size of an incoming request, but it is
405        // an operator-configurable value. Concurrency is capped independently via
406        // `MAX_CONCURRENT_FUTURES` so that increasing the request limit does not
407        // proportionally increase the number of concurrent BigTable calls per request.
408        let max_buffered_concurrent_futures = req.len().clamp(1, MAX_CONCURRENT_FUTURES);
409
410        stream::iter(req.into_iter())
411            .map(|range| self.object_before_version(range))
412            .buffered(max_buffered_concurrent_futures)
413            .try_collect::<Vec<Option<Bytes>>>()
414            .await
415    }
416}

418/// Extracts specific key type from a general [`Key`] type.
419///
420/// Takes:
421/// - `keys`: The list of keys to extract from
422/// - `extractor`: Function that returns Some(extracted_value) for the target
423///   variant, None otherwise
424///
425/// Returns a vector of extracted values. Returns [`ApiError::BadRequest`]
426/// if any extraction returns None value.
427fn extract_keys<T, F>(keys: &[Key], extractor: F) -> Result<Vec<T>, ApiError>
428where
429    F: Fn(&Key) -> Option<T>,
430{
431    keys.iter()
432        .map(|k| {
433            extractor(k).ok_or_else(|| {
434                ApiError::BadRequest("all keys should be of the same type".to_string())
435            })
436        })
437        .collect()
438}

440/// Represents a range of keys for a single object the BigTableDB is allowed to
441/// scan.
442pub(crate) struct ObjectRangeKeyBound {
443    start_key: StartKey,
444    end_key: EndKey,
445}
446
447impl ObjectRangeKeyBound {
448    /// Returns `true` if the range contains no rows.
449    fn is_empty(&self) -> bool {
450        match (&self.start_key, &self.end_key) {
451            (StartKey::StartKeyClosed(start), EndKey::EndKeyClosed(end)) => start > end,
452            (
453                StartKey::StartKeyClosed(start) | StartKey::StartKeyOpen(start),
454                EndKey::EndKeyClosed(end) | EndKey::EndKeyOpen(end),
455            ) => start >= end,
456        }
457    }
458}
459
460impl TryFrom<Key> for ObjectRangeKeyBound {
461    type Error = RangeKeyBoundError;
462
463    fn try_from(key: Key) -> Result<Self, Self::Error> {
464        let Key::ObjectKey(object_key) = key else {
465            return Err(RangeKeyBoundError::UnexpectedItemType {
466                expected: ItemType::Object,
467                detail: "range key must be an ObjectKey".into(),
468            });
469        };
470        Ok(object_key.into())
471    }
472}
473
474impl From<ObjectKey> for ObjectRangeKeyBound {
475    fn from(value: ObjectKey) -> Self {
476        let obj_id = value.0;
477        ObjectRangeKeyBound {
478            start_key: StartKey::StartKeyClosed(raw_object_key(&ObjectKey::min_for_id(&obj_id))),
479            end_key: EndKey::EndKeyOpen(raw_object_key(&value)),
480        }
481    }
482}

484/// Represents a request to fetch objects before a given version.
485///
486/// This struct expects that all keys in the input vector are of type
487/// [`Key::ObjectKey`] in the [`TryFrom`] implementation.
488pub(crate) struct ObjectsBeforeVersionRequest(Vec<ObjectRangeKeyBound>);
489
490impl ObjectsBeforeVersionRequest {
491    fn into_inner(self) -> Vec<ObjectRangeKeyBound> {
492        self.0
493    }
494}
495
496impl TryFrom<Vec<Key>> for ObjectsBeforeVersionRequest {
497    type Error = RangeKeyBoundError;
498
499    fn try_from(keys: Vec<Key>) -> Result<Self, Self::Error> {
500        let object_range_keys = extract_keys(&keys, |k| match k {
501            Key::ObjectKey(object_key) => Some(*object_key),
502            _ => None,
503        })
504        .map_err(|e| RangeKeyBoundError::UnexpectedItemType {
505            expected: ItemType::Object,
506            detail: e.to_string(),
507        })?
508        .into_iter()
509        .map(ObjectRangeKeyBound::from)
510        .collect();
511
512        Ok(ObjectsBeforeVersionRequest(object_range_keys))
513    }
