20const BEFORE_VERSION_REQUIRES_OB_ERROR_MSG: &str =
21    "`before_version` query parameter is only valid for `ob` item types";
22
23/// Request payload for multi_get_objects_post containing list of keys.
24#[derive(Deserialize, Debug)]
25pub(crate) struct MultiGetRequest {
26    /// List of base64url-encoded keys to retrieve.
27    pub(crate) keys: Vec<String>,
28}

30/// Extracts the `?before_version` query parameter
31#[derive(Deserialize, Debug, Default)]
32pub(crate) struct BeforeVersion {
33    #[serde(default)]
34    pub(crate) before_version: bool,
35}

37/// Retrieves data associated with a given key from the KV store as raw
38/// [`Bytes`](bytes::Bytes).
39///
40/// # Query Parameters
41///
42/// * `before_version` (optional, default `false`): only valid when `item_type`
43///   is [`ItemType::Object`]. When `true`, returns the latest stored version
44///   strictly less than the version encoded in the key. Returns `400 Bad
45///   Request` if used with any other [`ItemType`].
46///
47/// # Returns
48///
49/// * If the key exists, the data is returned as a [`Bytes`](bytes::Bytes)
50///   stream with a `200 OK` status code.
51/// * If the key does not exist, a `404 Not Found` status code is returned with
52///   an empty body.
53/// * If an error occurs while interacting with the KV store, an `500 internal
54///   server error` is returned.
55pub async fn data_as_bytes(
56    State(app_state): State<SharedRestServerAppState>,
57    ExtractPath(key): ExtractPath,
58    Query(BeforeVersion { before_version }): Query<BeforeVersion>,
59) -> Result<impl IntoResponse, ApiError> {
60    if before_version {
61        let range = ObjectRangeKeyBound::try_from(key)
62            .map_err(|_| ApiError::BadRequest(BEFORE_VERSION_REQUIRES_OB_ERROR_MSG.into()))?;
63
64        let response = app_state
65            .kv_store_client
66            .object_before_version(range)
67            .await?;
68
69        return Ok(response.map_or_else(
70            || (StatusCode::NOT_FOUND, Body::empty()).into_response(),
71            |bytes| bytes.into_response(),
72        ));
73    }
74
75    app_state
76        .kv_store_client
77        .get(key)
78        .await
79        .map(|res| match res {
80            Some(bytes) => bytes.into_response(),
81            None => (StatusCode::NOT_FOUND, Body::empty()).into_response(),
82        })
83}

85/// Retrieves multiple objects via POST request with JSON payload.
86///
87/// # Path Parameters
88///
89/// - `item_type`: The type of items to get (e.g., "cs", "cc", "tx")
90///
91/// # Query Parameters
92///
93/// * `before_version` (optional, default `false`): only valid when `item_type`
94///   is [`ItemType::Object`]. When `true`, returns the latest stored version
95///   strictly less than the version encoded in each key. Returns `400 Bad
96///   Request` if used with any other [`ItemType`].
97///
98/// # Request Body
99///
100/// JSON object with `keys` field:
101///
102/// ```json
103/// {
104///   "keys": ["AAEAAAAAAAAA", "AAIAAAAAAAAA", "AAMAAAAAAAAA"]
105/// }
106/// ```
107///
108/// Where:
109/// - `keys`: Array of base64url-encoded keys for given `item_type`. The same
110///   kind of key and encoding user would use in single item GET request.
111///
112/// # Returns
113///
114/// * If successful, returns a BCS-serialized
115///   [`Vec`]<[`Option`]<[`Bytes`](bytes::Bytes)>> with a `200 OK` status code.
116///   The vector has the same length and order as the `keys` list in the request
117///   body. Each entry is `Some(bytes)` if the key was found, or `None` if the
118///   key was not found.
119///  * If no keys are provided or the number of keys exceeds the configured
120///    `multiget_max_items` limit, a `400 bad request error` is returned.
121/// * If the keys cannot be parsed, a `400 bad request error` is returned.
122/// * If an error occurs while interacting with the KV store, an `500 internal
123///   server error` is returned.
124pub async fn multi_get_data(
125    State(app_state): State<SharedRestServerAppState>,
126    Path(item_type): Path<ItemType>,
127    Query(BeforeVersion { before_version }): Query<BeforeVersion>,
128    Json(payload): Json<MultiGetRequest>,
129) -> Result<impl IntoResponse, ApiError> {
130    if payload.keys.is_empty() {
131        return Err(ApiError::BadRequest("no keys provided".into()));
132    }
133
134    if payload.keys.len() > app_state.multiget_max_items.get() {
135        return Err(ApiError::BadRequest(format!(
136            "too many keys: requested {}, maximum allowed is {}",
137            payload.keys.len(),
138            app_state.multiget_max_items
139        )));
140    }
141
142    let item_type_str = item_type.to_string();
143    let keys = payload
144        .keys
145        .iter()
146        .map(|encoded_key| {
147            Key::new(item_type_str.as_str(), encoded_key.as_str())
148                .map_err(|err| ApiError::BadRequest(format!("invalid key '{encoded_key}': {err}")))
149        })
150        .collect::<Result<Vec<Key>, ApiError>>()?;
151
152    let results = if before_version {
153        let request = ObjectsBeforeVersionRequest::try_from(keys)
154            .map_err(|_| ApiError::BadRequest(BEFORE_VERSION_REQUIRES_OB_ERROR_MSG.into()))?;
155        app_state
156            .kv_store_client
157            .objects_before_version(request)
158            .await?
159    } else {
160        app_state.kv_store_client.get_items(keys).await?
161    };
162
163    let bcs_data = bcs::to_bytes(&results).map_err(|_| ApiError::InternalServerError)?;
164    Ok(bcs_data.into_response())
165}
