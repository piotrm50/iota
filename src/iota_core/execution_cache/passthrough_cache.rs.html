281impl ExecutionCacheWrite for PassthroughCache {
282    #[instrument(level = "debug", skip_all)]
283    fn try_write_transaction_outputs(
284        &self,
285        epoch_id: EpochId,
286        tx_outputs: Arc<TransactionOutputs>,
287    ) -> IotaResult {
288        let tx_digest = *tx_outputs.transaction.digest();
289        let effects_digest = tx_outputs.effects.digest();
290
291        // NOTE: We just check here that live markers exist, not that they are locked to
292        // a specific TX. Why?
293        // 1. Live markers existence prevents re-execution of old certs when objects
294        //    have been upgraded
295        // 2. Not all validators lock, just 2f+1, so transaction should proceed
296        //    regardless (But the live markers should exist which means previous
297        //    transactions finished)
298        // 3. Equivocation possible (different TX) but as long as 2f+1 approves current
299        //    TX its fine
300        // 4. Live markers may have existed when we started processing this tx, but
301        //    could have since been deleted by a concurrent tx that finished first. In
302        //    that case, check if the tx effects exist.
303        self.store
304            .check_owned_objects_are_live(&tx_outputs.live_object_markers_to_delete)?;
305
306        let batch = self
307            .store
308            .build_db_batch(epoch_id, std::slice::from_ref(&tx_outputs))?;
309        batch.write()?;
310
311        self.executed_effects_digests_notify_read
312            .notify(&tx_digest, &effects_digest);
313
314        for object in tx_outputs.written.values() {
315            super::notify_object_written(&self.object_notify_read, object);
316        }
317        for (object_key, marker_value) in &tx_outputs.markers {
318            super::notify_marker_written(&self.object_notify_read, object_key, marker_value);
319        }
320
321        self.metrics
322            .pending_notify_read
323            .set(self.executed_effects_digests_notify_read.num_pending() as i64);
324
325        Ok(())
326    }
327
328    fn try_acquire_transaction_locks<'a>(
329        &'a self,
330        epoch_store: &'a AuthorityPerEpochStore,
331        owned_input_objects: &'a [ObjectRef],
332        transaction: VerifiedSignedTransaction,
333    ) -> IotaResult {
334        self.store
335            .acquire_transaction_locks(epoch_store, owned_input_objects, transaction)
336    }

422#[cfg(test)]
423impl PassthroughCache {
424    pub(super) fn write_object_for_testing(&self, object: Object) {
425        use crate::authority::authority_store_types::get_store_object;
426
427        let object_ref = object.compute_object_reference();
428        let object_key = ObjectKey::from(object_ref);
429        let store_object = get_store_object(object.clone());
430        self.store
431            .perpetual_tables
432            .objects
433            .insert(&object_key, &store_object)
434            .unwrap();
435
436        super::notify_object_written(&self.object_notify_read, &object);
437    }
438
439    pub(super) fn write_marker_for_testing(
440        &self,
441        epoch_id: EpochId,
442        object_key: &ObjectKey,
443        marker_value: MarkerValue,
444    ) {
445        self.store
446            .perpetual_tables
447            .object_per_epoch_marker_table
448            .insert(&(epoch_id, *object_key), &marker_value)
449            .unwrap();
450
451        super::notify_marker_written(&self.object_notify_read, object_key, &marker_value);
452    }
