5use std::{collections::BTreeMap, fmt::Display};
6
7use iota_types::{digests::ConsensusCommitDigest, messages_consensus::ConsensusTransaction};
8
9use crate::consensus_types::AuthorityIndex;
10/// A list of tuples of:
11/// (block origin authority index, all transactions contained in the block).
12/// For each transaction, returns deserialized transaction and its serialized
13/// size.
14type ConsensusOutputTransactions = Vec<(AuthorityIndex, Vec<(ConsensusTransaction, usize)>)>;
15
16/// Per-authority misbehavior counts observed by consensus in a single commit.
17/// A bridge type between Starfish's internal observation state and IOTA's
18/// wire format: the trait impl on `starfish_core::CommittedSubDag` is
19/// responsible for extracting the counts; `observations_from_consensus_output`
20/// projects this struct onto whichever `MisbehaviorObservationsVN` is active
21/// for the current protocol version.
22///
23/// Per-field empty `Vec`s mean "not wired yet" and are zero-filled by the
24/// mapper. Today this struct is structurally identical to
25/// `MisbehaviorObservationsV1`; the separation exists so consensus's
26/// observation set and the wire schema can evolve on independent cadences.
27#[derive(Debug, Default)]
28pub struct ConsensusOutputMisbehaviorCounts {
29    pub faulty_blocks_provable: Vec<u64>,
30    pub faulty_blocks_unprovable: Vec<u64>,
31    pub missing_proposals: Vec<u64>,
32    pub equivocations: Vec<u64>,
33}
34
35pub(crate) trait ConsensusOutputAPI: Display {
36    fn reputation_score_sorted_desc(&self) -> Option<Vec<(AuthorityIndex, u64)>>;
37    fn leader_round(&self) -> u64;
38    fn leader_author_index(&self) -> AuthorityIndex;
39
40    /// Returns epoch UNIX timestamp in milliseconds
41    fn commit_timestamp_ms(&self) -> u64;
42
43    /// Returns a unique global index for each committed sub-dag.
44    fn commit_sub_dag_index(&self) -> u64;
45
46    /// Returns all transactions in the commit.
47    fn transactions(&self) -> ConsensusOutputTransactions;
48
49    /// Returns the digest of consensus output.
50    fn consensus_digest(&self) -> ConsensusCommitDigest;
51
52    fn number_of_headers_in_commit_by_authority(&self) -> Vec<(AuthorityIndex, u64)>;
53
54    /// Per-authority misbehavior counts observed in this commit. The default
55    /// impl returns the all-empty struct so unwired implementations get
56    /// correct behavior for free; the mapper zero-fills empty fields.
57    fn misbehavior_counts(&self) -> ConsensusOutputMisbehaviorCounts {
58        ConsensusOutputMisbehaviorCounts::default()
59    }
60}
61impl ConsensusOutputAPI for starfish_core::CommittedSubDag {
62    fn reputation_score_sorted_desc(&self) -> Option<Vec<(AuthorityIndex, u64)>> {
63        if !self.reputation_scores_desc.is_empty() {
64            Some(
65                self.reputation_scores_desc
66                    .iter()
67                    .map(|(id, score)| (id.value() as AuthorityIndex, *score))
68                    .collect(),
69            )
70        } else {
71            None
72        }
73    }
74
75    fn leader_round(&self) -> u64 {
76        self.leader.round as u64
77    }
78
79    fn leader_author_index(&self) -> AuthorityIndex {
80        self.leader.author.value() as AuthorityIndex
81    }
82
83    fn commit_timestamp_ms(&self) -> u64 {
84        self.timestamp_ms
85    }
86
87    fn commit_sub_dag_index(&self) -> u64 {
88        self.commit_ref.index.into()
89    }
90
91    fn transactions(&self) -> ConsensusOutputTransactions {
92        self.transactions
93            .iter()
94            .map(|vt| {
95                let round = vt.round() as u64;
96                let author = vt.author().value() as AuthorityIndex;
97
98                let transactions: Vec<_> = vt
99                    .transactions()
100                    .iter()
101                    .flat_map(|tx| {
102                        let transaction = bcs::from_bytes::<ConsensusTransaction>(tx.data());
103                        match transaction {
104                            Ok(transaction) => Some((transaction, tx.data().len())),
105                            Err(err) => {
106                                tracing::error!(
107                                    "Failed to deserialize sequenced consensus transaction \
108                                     (this should not happen) {err} from {author} at {round}"
109                                );
110                                None
111                            }
112                        }
113                    })
114                    .collect();
115
116                (author, transactions)
117            })
118            .collect()
119    }
120
121    fn consensus_digest(&self) -> ConsensusCommitDigest {
122        // Ensure wire layout matches.
123        static_assertions::assert_eq_size!(ConsensusCommitDigest, starfish_core::CommitDigest);
124        ConsensusCommitDigest::new(self.commit_ref.digest.into_inner())
125    }
126
127    fn number_of_headers_in_commit_by_authority(&self) -> Vec<(AuthorityIndex, u64)> {
128        let mut num_of_committed_headers = BTreeMap::new();
129        self.base
130            .committed_header_refs
131            .iter()
132            .for_each(|block_ref| {
133                let author_index = block_ref.author.value() as AuthorityIndex;
134                *num_of_committed_headers.entry(author_index).or_insert(0) += 1;
135            });
136        num_of_committed_headers.into_iter().collect()
137    }
138}
